Overview

Exploratory data analysis of 1,472 Sephora cosmetics products using R. Dataset includes product category, brand, price, rating, and skin type suitability.

Products by Category

category_counts <- table(cosmetics$Label)
barplot(category_counts, main="Number of Products by Category",
        col="pink", las=2, cex.names=0.7)

Average Price by Category

avg_price <- round(tapply(cosmetics$Price, cosmetics$Label, mean), 2)
barplot(sort(avg_price, decreasing=TRUE), main="Average Price by Category",
        col="lightcoral", las=2, cex.names=0.7)

## Price Distribution

hist(cosmetics$Price, main="Distribution of Product Prices",
     xlab="Price (USD)", col="lavender", breaks=20)

Skin Type Suitability

skin_types <- c("Combination","Dry","Normal","Oily","Sensitive")
skin_counts <- colSums(cosmetics[, skin_types], na.rm=TRUE)
barplot(skin_counts, main="Products per Skin Type",
        col=c("peachpuff","lightblue","lightgreen","lightyellow","plum"))
